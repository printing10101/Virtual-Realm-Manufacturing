# TDengine 时序数据库 - M0.2 任务新增
# ---------------------------------------------------------------------------
# 添加原因：
#   APT 系统需存储机床高频时序数据（主轴转速/振动/温度等），现有 PostgreSQL
#   在此类场景下存在性能瓶颈。TDengine 时序数据库针对高频写入与时间范围
#   查询做了专门优化，可显著降低单条记录写入延迟并支持毫秒级聚合。
# 选型说明：
#   - ``taospy`` 是官方 Python 驱动，支持 native 协议（端口 6030），
#     单连接即可承担数万 QPS 的写入负载。
#   - 备选 ``taos-ws-py``（WebSocket 协议）适用于跨网络边界场景。
# 版本约束：与根 requirements.txt 保持一致。
# ---------------------------------------------------------------------------
taospy==2.7.0            # TDengine 时序数据库官方 Python 驱动

# MTConnect 适配器 - M0.3 任务新增
# ---------------------------------------------------------------------------
# 添加原因：
#   APT 数据底座需从 MTConnect Agent 周期化拉取机床运行参数（主轴转速、
#   负载、进给速度、执行状态等），并写入 TDengine 时序库。``requests`` 用
#   于与 MTConnect Agent 建立 HTTP/HTTPS 长连接、``lxml`` 作为高性能 XML
#   解析器在 M0.3 后续迭代中替换 stdlib 的 ``ElementTree``（响应体可能
#   达到数 MB）。
# 选型说明：
#   - ``requests`` 是 Python 生态中事实标准的同步 HTTP 客户端，与本任务
#     的 CLI / 短轮询场景天然契合，APM 与抓包工具也最完善。
#   - ``lxml`` 比 ``ElementTree`` 更快、对大文档更鲁棒，后续 M0.3+ 增加
#     /assets 端点（描述文件较大）时可以直接切换。
# 版本约束：与根 requirements.txt 保持一致。
# ---------------------------------------------------------------------------
requests==2.32.3         # MTConnect Agent HTTP 客户端
lxml==5.3.0              # 高速 XML 解析（stdlib fallback 已就绪）

# OPC UA 适配器 - M5.4 任务新增
# ---------------------------------------------------------------------------
# 添加原因：
#   APT 数据底座需从 OPC UA 服务器实时订阅机床运行参数（主轴转速、负载、
#   进给速度、执行状态等），并写入 TDengine 时序库。``asyncua`` 是纯 Python
#   实现的 OPC UA 客户端/服务器库，支持异步订阅机制，与本项目的异步架构
#   天然契合。
# 选型说明：
#   - ``asyncua`` 是 Python 生态中最成熟的 OPC UA 实现，支持完整的 OPC UA
#     协议栈，包括节点订阅、数据变更通知等功能。
#   - 异步设计使其能够高效处理高频数据流，与 TDengine 批写机制配合良好。
# 版本约束：与根 requirements.txt 保持一致。
# ---------------------------------------------------------------------------
asyncua==1.1.5           # OPC UA 客户端/服务器库（异步订阅支持）
