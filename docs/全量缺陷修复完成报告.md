import numpy as np
from numba import jit

# 使用 NumPy 向量化操作替代三重嵌套循环
@jit(nopython=True)
def generate_tool_mask_vectorized(grid_shape, tool_path, tool_radius):
    """向量化的刀具掩码生成"""
    x, y, z = np.ogrid[0:grid_shape[0], 0:grid_shape[1], 0:grid_shape[2]]
    
    # 计算到刀具路径的距离
    distance = np.sqrt(
        (x - tool_path[0])**2 + 
        (y - tool_path[1])**2 + 
        (z - tool_path[2])**2
    )
    
    # 生成掩码
    mask = distance <= tool_radius
    return mask

# 使用 Marching Cubes 算法重建网格
from skimage.measure import marching_cubes

def rebuild_mesh(voxels):
    """使用 Marching Cubes 重建网格"""
    verts, faces, normals, values = marching_cubes(voxels, level=0.5)
    return verts, faces
